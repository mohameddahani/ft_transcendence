1. ACCESS TOKEN (SHORT LIFE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Purpose:
- Authenticate user quickly
- Expire fast if stolen

Example:

    jwt.sign(payload, {
        expiresIn: '10m'
    })

Best Practice:
✓ Store in HttpOnly cookie
✓ Secure=true
✓ SameSite=strict


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. REFRESH TOKEN (LONGER LIFE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Purpose:
- Generate new access tokens safely

Generate:

    const refreshToken = jwt.sign(payload, {
        expiresIn: '7d'
    });

Hash before database storage:

    const hash = await argon2.hash(refreshToken);

Store:

    {
        userId,
        tokenHash: hash
    }

Never store raw refresh token.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. SESSION BINDING (DEVICE PROTECTION)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Store:

    {
        sessionId,
        userId,
        ip,
        userAgent,
        deviceFingerprint
    }

Validation:

    if (
        session.ip !== req.ip ||
        session.userAgent !== req.headers['user-agent']
    ) {
        throw new UnauthorizedException();
    }

Result:
If friend copies token → blocked.

4. ROTATE REFRESH TOKENS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Flow:

Login:
    refresh_1

Refresh:
    refresh_1 → invalidate
    issue refresh_2

If attacker later uses refresh_1:
→ Security breach detected
→ Revoke all sessions
